-- PostgreSQL 16.15, practice data: 5 locations, 2,000 users, 100,000 tickets (200 WAITING).
-- Measured in the Claude sandbox - your numbers will differ, the plan shapes should be similar.
-- FIFO waiting tickets of one location - WITH index (locationId, status) [as in schema.prisma]

EXPLAIN (ANALYZE, BUFFERS)
SELECT t."ticketNumber", u."name", t."createdAt"
FROM "QueueTicket" t JOIN "User" u ON u."id" = t."userId"
WHERE t."locationId" = 1 AND t."status" = 'WAITING'
ORDER BY t."ticketNumber" ASC;
